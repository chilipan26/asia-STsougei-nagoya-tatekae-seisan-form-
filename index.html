/** * 立替金精算システム – 納品用バックエンド (GAS) * ============================================================ * 【使い方】 * 1. このスクリプトを対象スプレッドシートのコンテナスクリプトとして開く * （またはスタンドアロンにして下記 SPREADSHEET_ID を設定） * 2. CONFIG.SPREADSHEET_ID を対象スプレッドシートのIDに変更 * 3. setupSpreadsheet() を実行 → シート構成・数式・チェックボックスを自動生成 * 4. 「デプロイ」→「新しいデプロイ」→ ウェブアプリ として公開 * （実行ユーザー：自分／アクセスできるユーザー：全員） * 5. 発行されたURLを、フォーム側HTMLの GAS_URL に設定 * ============================================================ */ const CONFIG = { SPREADSHEET_ID: '★ここにスプレッドシートIDを入力してください', SHEET_LOG: '申請ログ', SHEET_PAYMENT: '支払いログ', SHEET_SUMMARY: '人別サマリー', // 領収書画像の保存先フォルダ（「大きな箱」）。 // 指定したGoogleドライブフォルダのIDを入れると、そこに画像がまとめて入ります。 // 空欄のままにすると、スクリプトが自動でフォルダを1つ作って使い回します。 DRIVE_FOLDER_ID: '1JiolNR5v2dhdQa_WfWCBC-ns6n7O3Bf5', DRIVE_FOLDER_PROP_KEY: 'RECEIPT_FOLDER_ID', CHECKBOX_PRESET_ROWS: 1000, // 事前にチェックボックス書式・入力規則を適用しておく行数 }; // ============================================================ // 受信エンドポイント（フォームからのPOSTを受け取る） // ============================================================ function doPost(e) { try { const payload = JSON.parse(e.postData.contents); if (payload.action !== 'submitReimbursement') { return jsonResponse({ status: 'error', message: 'invalid action' }); } const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); const logSheet = ss.getSheetByName(CONFIG.SHEET_LOG); if (!logSheet) { throw new Error('シート「' + CONFIG.SHEET_LOG + '」が見つかりません。先に setupSpreadsheet() を実行してください。'); } const staffName = String(payload.staffName || '').trim(); const timestamp = payload.timestamp ? new Date(payload.timestamp) : new Date(); const items = payload.items || []; if (!staffName) return jsonResponse({ status: 'error', message: '氏名が未入力です' }); if (!items.length) return jsonResponse({ status: 'error', message: '申請項目がありません' }); const folder = getReceiptFolder(); const uploadedImages = {}; // 同じ画像（同一カード内の複数行）は1回だけアップロードして使い回す items.forEach(function (item) { let imageUrl = ''; if (item.fileData) { if (uploadedImages[item.fileData]) { imageUrl = uploadedImages[item.fileData]; } else { imageUrl = saveReceiptImage( folder, item.fileData, item.mimeType || 'image/jpeg', staffName, item.date, item.vehicle ); uploadedImages[item.fileData] = imageUrl; } } const row = logSheet.getLastRow() + 1; logSheet.getRange(row, 1, 1, 8).setValues([[ timestamp, staffName, item.vehicle || '', item.date || '', item.purpose || '', item.route || '', Number(item.amount) || 0, imageUrl, ]]); // I列（レシート受領チェック）は setupSpreadsheet() で事前に // チェックボックス書式を適用済みのため、値は未設定＝未チェックのまま }); return jsonResponse({ status: 'ok' }); } catch (err) { return jsonResponse({ status: 'error', message: String(err) }); } } function jsonResponse(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } // ============================================================ // 領収書画像をDriveに保存してURLを返す // ファイル名は「日付 車両番号 氏名.jpg」（同名が既にあれば _2, _3... を付与） // ============================================================ function saveReceiptImage(folder, base64Data, mimeType, staffName, dateStr, vehicle) { const baseName = [dateStr || '', vehicle || '', staffName || ''] .filter(function (s) { return s; }) .join(' '); const ext = mimeType === 'application/pdf' ? 'pdf' : 'jpg'; let fileName = baseName + '.' + ext; let counter = 2; while (folder.getFilesByName(fileName).hasNext()) { fileName = baseName + '_' + counter + '.' + ext; counter++; } const blob = Utilities.newBlob(Utilities.base64Decode(base64Data), mimeType, fileName); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); return file.getUrl(); } // 保存先フォルダ（「大きな箱」）を取得。 // CONFIG.DRIVE_FOLDER_ID が設定されていればそのフォルダを使用し、 // 未設定ならスクリプトプロパティに記憶した自動作成フォルダを使い回す。 function getReceiptFolder() { if (CONFIG.DRIVE_FOLDER_ID && CONFIG.DRIVE_FOLDER_ID.indexOf('★') !== 0) { return DriveApp.getFolderById(CONFIG.DRIVE_FOLDER_ID); } return getOrCreateReceiptFolder(); } function getOrCreateReceiptFolder() { const props = PropertiesService.getScriptProperties(); const existingId = props.getProperty(CONFIG.DRIVE_FOLDER_PROP_KEY); if (existingId) { try { return DriveApp.getFolderById(existingId); } catch (e) { // フォルダが削除されていた場合は作り直す } } const folder = DriveApp.createFolder('立替精算_領収書画像'); props.setProperty(CONFIG.DRIVE_FOLDER_PROP_KEY, folder.getId()); return folder; } // ============================================================ // 初回セットアップ：シート構成・書式・数式を自動生成 // 新しい案件に転用するときは、CONFIG.SPREADSHEET_ID を変更してから // この関数を1回実行するだけでOK // ============================================================ function setupSpreadsheet() { const ss = SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID); // ---- 申請ログ ---- let logSheet = ss.getSheetByName(CONFIG.SHEET_LOG); if (!logSheet) logSheet = ss.insertSheet(CONFIG.SHEET_LOG); logSheet.clear(); logSheet.getRange(1, 1, 1, 9).setValues([[ 'タイムスタンプ', '申請者名', '車両番号', '勤務日', '用途', '区間', '金額', '領収書画像URL', 'レシート受領チェック（メモ用）', ]]); logSheet.getRange(1, 1, 1, 9).setFontWeight('bold'); logSheet.getRange(2, 9, CONFIG.CHECKBOX_PRESET_ROWS, 1).insertCheckboxes(); logSheet.setFrozenRows(1); // ---- 支払いログ ---- let paymentSheet = ss.getSheetByName(CONFIG.SHEET_PAYMENT); if (!paymentSheet) paymentSheet = ss.insertSheet(CONFIG.SHEET_PAYMENT); paymentSheet.clear(); paymentSheet.getRange(1, 1, 1, 4).setValues([[ '支払日', '支払対象者', '支払金額', '備考', ]]); paymentSheet.getRange(1, 1, 1, 4).setFontWeight('bold'); paymentSheet.setFrozenRows(1); // 支払対象者は申請ログの申請者名から選択（入力ミス防止・完全一致必須ではない＝警告のみ） const nameListRange = logSheet.getRange('B2:B' + CONFIG.CHECKBOX_PRESET_ROWS); const rule = SpreadsheetApp.newDataValidation() .requireValueInRange(nameListRange, true) .setAllowInvalid(true) .build(); paymentSheet.getRange(2, 2, CONFIG.CHECKBOX_PRESET_ROWS, 1).setDataValidation(rule); // ---- 人別サマリー（自動集計・新規申請者も自動で追加される）---- let summarySheet = ss.getSheetByName(CONFIG.SHEET_SUMMARY); if (!summarySheet) summarySheet = ss.insertSheet(CONFIG.SHEET_SUMMARY); summarySheet.clear(); summarySheet.getRange(1, 1, 1, 4).setValues([[ '申請者名', '累計申請額', '支払済み累計', '未精算残高', ]]); summarySheet.getRange(1, 1, 1, 4).setFontWeight('bold'); summarySheet.setFrozenRows(1); summarySheet.getRange('A2').setFormula( '=SORT(UNIQUE(FILTER(' + CONFIG.SHEET_LOG + '!B2:B,' + CONFIG.SHEET_LOG + '!B2:B<>"")))' ); summarySheet.getRange('B2').setFormula( '=IFERROR(ARRAYFORMULA(IF(A2:A="","",SUMIF(' + CONFIG.SHEET_LOG + '!B2:B,A2:A,' + CONFIG.SHEET_LOG + '!G2:G))),"")' ); summarySheet.getRange('C2').setFormula( '=IFERROR(ARRAYFORMULA(IF(A2:A="","",SUMIF(' + CONFIG.SHEET_PAYMENT + '!B2:B,A2:A,' + CONFIG.SHEET_PAYMENT + '!C2:C))),"")' ); summarySheet.getRange('D2').setFormula( '=ARRAYFORMULA(IF(A2:A="","",B2:B-C2:C))' ); SpreadsheetApp.flush(); Logger.log('✅ セットアップ完了：申請ログ／支払いログ／人別サマリー を作成しました。'); }
